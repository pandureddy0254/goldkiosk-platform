using System;
using System.Linq;
using System.Text.Json;
using System.Text.Json.Serialization;
using NSec.Cryptography;

namespace GoldKiosk.Admin.Licensing;

public class License
{
    [JsonPropertyName("iss")] public string Issuer       { get; set; } = "";
    [JsonPropertyName("kid")] public string Kid          { get; set; } = "";
    [JsonPropertyName("ver")] public int    Version      { get; set; } = 1;
    [JsonPropertyName("tid")] public Guid   TenantId     { get; set; }
    [JsonPropertyName("pid")] public Guid   PartnerId    { get; set; }
    [JsonPropertyName("co")]  public string LegalName    { get; set; } = "";
    [JsonPropertyName("dn")]  public string DisplayName  { get; set; } = "";
    [JsonPropertyName("pl")]  public string PlanCode     { get; set; } = "starter";
    [JsonPropertyName("ft")]  public string[] Features   { get; set; } = Array.Empty<string>();
    [JsonPropertyName("kc")]  public int    KioskCap     { get; set; }
    [JsonPropertyName("rg")]  public string RegionHint   { get; set; } = "";
    [JsonPropertyName("eml")] public string AdminEmail   { get; set; } = "";
    [JsonPropertyName("iat")] public long   IssuedAt     { get; set; }
    [JsonPropertyName("exp")] public long   ExpiresAt    { get; set; }

    public DateTimeOffset ExpiresOn => DateTimeOffset.FromUnixTimeSeconds(ExpiresAt);
    public bool HasFeature(string code) => Features.Contains(code, StringComparer.OrdinalIgnoreCase);
}

public record VerifyOutcome(bool Valid, License? License, string? Error);

public class LicenseVerifier
{
    // kid → 32-byte Ed25519 public key, base64. Populate from JWKS at startup
    // or pin in config. Multiple entries support key rotation.
    private readonly Dictionary<string, byte[]> _publicKeysByKid;

    public LicenseVerifier(Dictionary<string, byte[]> publicKeysByKid) =>
        _publicKeysByKid = publicKeysByKid;

    /// <summary>
    /// Build from a JWKS JSON document fetched from
    /// https://crm.goldkiosk.ai/.well-known/jwks.json.
    /// </summary>
    public static LicenseVerifier FromJwks(string jwksJson)
    {
        using var doc = JsonDocument.Parse(jwksJson);
        var map = new Dictionary<string, byte[]>();
        foreach (var k in doc.RootElement.GetProperty("keys").EnumerateArray())
        {
            if (k.GetProperty("kty").GetString() != "OKP") continue;
            if (k.GetProperty("crv").GetString() != "Ed25519") continue;
            var kid = k.GetProperty("kid").GetString() ?? "";
            var x   = k.GetProperty("x").GetString() ?? "";
            map[kid] = Base64UrlDecode(x);
        }
        return new LicenseVerifier(map);
    }

    public VerifyOutcome Verify(string token)
    {
        const string prefix = "AIKI-";
        if (string.IsNullOrWhiteSpace(token) || !token.StartsWith(prefix))
            return new(false, null, "Token must start with AIKI-.");
        var body = token[prefix.Length..];
        var dot = body.IndexOf('.');
        if (dot <= 0 || dot == body.Length - 1)
            return new(false, null, "Token is malformed (missing '.').");

        byte[] payloadBytes, sig;
        try
        {
            payloadBytes = Base64UrlDecode(body[..dot]);
            sig          = Base64UrlDecode(body[(dot + 1)..]);
        }
        catch (FormatException) { return new(false, null, "Token contains invalid base64url."); }

        License? license;
        try { license = JsonSerializer.Deserialize<License>(payloadBytes); }
        catch (JsonException ex) { return new(false, null, "Token payload is not JSON: " + ex.Message); }
        if (license is null) return new(false, null, "Token payload deserialized to null.");

        if (!_publicKeysByKid.TryGetValue(license.Kid, out var pkRaw))
            return new(false, license, $"Unknown signing key id '{license.Kid}'.");

        var pk = PublicKey.Import(SignatureAlgorithm.Ed25519, pkRaw, KeyBlobFormat.RawPublicKey);
        if (!SignatureAlgorithm.Ed25519.Verify(pk, payloadBytes, sig))
            return new(false, license, "Signature does not match payload.");

        var now = DateTimeOffset.UtcNow.ToUnixTimeSeconds();
        if (license.ExpiresAt > 0 && license.ExpiresAt < now)
            return new(false, license, "License expired " +
                DateTimeOffset.FromUnixTimeSeconds(license.ExpiresAt).ToString("u") + ".");

        return new(true, license, null);
    }

    private static byte[] Base64UrlDecode(string s)
    {
        var padded = s.Replace('-', '+').Replace('_', '/');
        var pad = padded.Length % 4;
        if (pad > 0) padded += new string('=', 4 - pad);
        return Convert.FromBase64String(padded);
    }
}
